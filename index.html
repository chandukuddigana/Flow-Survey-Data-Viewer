function processRainFile(text, fileName) { const lines = text.split(/\r?\n/); let intervalMinutes = 2; // Default let rawIntensity = []; // Values as mm/hr let inData = false; // 1. Extract Interval and Data for (let line of lines) { if (line.match(/\s+(\d+)\s*$/)) { // Logic to capture interval from your header } if (line.startsWith('*CEND')) { inData = true; continue; } if (inData) { line.split(/\s+/).forEach(val => { let v = parseFloat(val); if (!isNaN(v)) rawIntensity.push(v); }); } } // 2. The Conversion Logic (Crucial for your requirements) const intervalHours = intervalMinutes / 60; const depths = rawIntensity.map(intensity => intensity * intervalHours); // Calculate total depth for the stats table const totalDepth = depths.reduce((sum, d) => sum + d, 0); return { intensity: rawIntensity, // Used for the "mm/hr" graph depths: depths, // Used for "Cumulative (mm)" graph totalDepth: totalDepth // Used for the stats table }; }
